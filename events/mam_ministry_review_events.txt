  save_temporary_scope_as = mam_mr_dispatch_actor

   AND = { mam_ministry_upper_authority_trigger = yes title:h_china = {
    exists = var:mam_mr_upper_open_file var:mam_mr_upper_open_actor = scope:mam_mr_dispatch_actor

    var:mam_mr_upper_open_file = { mam_ministry_review_current_trigger = yes }

   AND = { mam_ministry_lower_authority_trigger = yes title:h_china = {
    exists = var:mam_mr_lower_open_file var:mam_mr_lower_open_actor = scope:mam_mr_dispatch_actor
    NOT = { has_variable = mam_mr_lower_presented }
    var:mam_mr_lower_open_file = { mam_ministry_review_current_trigger = yes }
   } }
  }
 }

  save_scope_as = mam_mr_actor save_scope_as = mam_minister
  if = { limit = { mam_ministry_upper_authority_trigger = yes }
   title:h_china = { set_variable = { name = mam_mr_upper_presented value = 1 } var:mam_mr_upper_open_file = { save_scope_as = mam_mr_file } }

  else = { title:h_china = { set_variable = { name = mam_mr_lower_presented value = 1 } var:mam_mr_lower_open_file = { save_scope_as = mam_mr_file } } }

  scope:mam_mr_file.var:mam_mr_predecessor = { save_scope_as = mam_mr_predecessor }

  scope:mam_mr_file.var:mam_mr_preferred ?= { save_scope_as = mam_mr_preferred }
  scope:mam_mr_file.var:mam_mr_marked ?= { save_scope_as = mam_mr_marked }
  scope:mam_mr_file.var:mam_mr_benchmark ?= { save_scope_as = mam_mr_benchmark }
