# One bounded callback per member; native helpers execute with this character as
# event root, preserving native eligibility/AI calculations and contribution credit.
mam_fund.31 = {
 type = character_event hidden = yes
 trigger = { mam_revenue_ministry_trigger = yes is_imprisoned = no NOT = { has_trait = incapable } }
 immediate = {
  save_scope_value_as = { name = mam_cf_manual_invoice_action value = 0 }
  mam_revenue_fund_auto_start_effect = yes
  if = { limit = { OR = { NOT = { has_variable = mam_pc_retry_invoice } var:mam_pc_retry_invoice <= current_date } }
   if = { limit = { has_variable = mam_pc_wanted_project NOT = { exists = var:mam_pc_wanted_project } }
    remove_variable = mam_pc_wanted_project remove_variable = mam_pc_wanted_key
   }
   mam_revenue_fund_find_project_invoice_effect = yes
   if = { limit = { exists = scope:mam_cf_pick_project }
    mam_revenue_fund_process_project_invoice_effect = yes
   }
   else_if = { limit = { has_variable = mam_pc_wanted_project }
    remove_variable = mam_pc_wanted_project remove_variable = mam_pc_wanted_key
   }
  }
 }
}
