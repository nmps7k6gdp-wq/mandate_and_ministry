mam_personnel.0010 = {
    type = character_event
    title = mam_personnel.0010.t
    desc = mam_personnel.0010.desc
    theme = merit
    left_portrait = { character = scope:mam_inspected animation = reading }
    immediate = {
        mam_resolve_register_authority_effect = yes
        scope:mam_inspected = {
            save_scope_value_as = { name = mam_detail_merit value = mam_merit_points }
            save_scope_value_as = { name = mam_detail_qualification value = mam_qualification_points }
            save_scope_value_as = { name = mam_detail_skills value = mam_skill_points }
            save_scope_value_as = { name = mam_detail_preference value = mam_minister_preference }

        save_scope_value_as = { name = mam_detail_distinction value = mam_exam_distinction_points }
            save_scope_value_as = { name = mam_detail_total value = mam_recommendation_score }
        }
    }
    option = { name = mam_personnel.close }

    option = {
        name = mam_credentials.back
        trigger = {
            exists = scope:mam_credentials_origin
            scope:mam_credentials_origin > 0
            trigger_if = {
                limit = { scope:mam_credentials_origin = 2 }
                exists = scope:mam_posting
            }
        }
        hidden_effect = {
            if = {
                limit = { scope:mam_credentials_origin = 2 }
                trigger_event = mam_personnel.0021
            }
            else = {
                mam_build_register_effect = yes
                if = {
                    limit = { exists = scope:mam_review_mode scope:mam_review_mode = 1 }
                    trigger_event = mam_personnel.0002
                }
                else = { trigger_event = mam_personnel.0001 }
            }
        }
    }

}
