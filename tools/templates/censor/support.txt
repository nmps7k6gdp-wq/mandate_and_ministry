mam_censor_inspection_story = {
 visible = no on_setup = { }
 on_owner_death = {
  clear_saved_scope = mam_cen_heir
  var:mam_cen_realm.holder ?= { if = { limit = { is_alive = yes } save_scope_as = mam_cen_heir } }
  if = { limit = { NOT = { exists = scope:mam_cen_heir } }
   var:mam_cen_realm.current_heir ?= { if = { limit = { is_alive = yes } save_scope_as = mam_cen_heir } }
  }
  if = { limit = { exists = scope:mam_cen_heir } make_story_owner = scope:mam_cen_heir }
 }
 effect_group = { days = 7 triggered_effect = {
  trigger = { always = yes }
  effect = {
   if = { limit = { var:mam_cen_pending = 1 }
    if = { limit = { mam_censor_live_trigger = yes } mam_censor_tick_effect = yes }
    else = {
     # An orphaned/de-indexed record must not live forever or sanction anyone.
     set_variable = { name = mam_cen_pending value = 0 }
     set_variable = { name = mam_cen_result value = 10 }
     set_variable = { name = mam_cen_closed value = current_date }
     change_variable = { name = mam_cen_serial add = 1 }
    }
   }
   else_if = { limit = { has_variable = mam_cen_closed var:mam_cen_closed <= { value = current_date subtract = 3650 } }
    end_story = yes
   }
  }
 } }
}
