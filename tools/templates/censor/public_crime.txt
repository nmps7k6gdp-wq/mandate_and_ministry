# Sovereign context, bounded to one new public referral per weekly intake.
mam_censor_public_poll_effect = {
 if = { limit = { mam_censor_emperor_trigger = yes }
  every_vassal_or_below = { limit = { mam_censor_public_subject_trigger = yes has_variable = mam_cen_public_seen
   is_imprisoned = no mam_censor_public_ground_trigger = no
  }
   # A genuinely cleared episode may later acquire new grounds. Never reopen
   # a pending inspection just because its arrest consumed the native reason.
   if = { limit = { NOT = { exists = var:mam_cen_public_record } }
    remove_variable = mam_cen_public_record remove_variable = mam_cen_public_seen
   }
   else_if = { limit = { var:mam_cen_public_record = { var:mam_cen_intel_pending = 0
    trigger_if = { limit = { exists = var:mam_cen_inspection } var:mam_cen_inspection = { mam_censor_live_trigger = no } }
   } }
    remove_variable = mam_cen_public_record remove_variable = mam_cen_public_seen
   }
  }
  if = { limit = { NOT = { title:h_china = { any_in_list = { variable = mam_cen_intel_queue count >= 8 always = yes } } }
   exists = title:e_minister_censor.holder
  }
   ordered_vassal_or_below = { limit = {
    mam_censor_public_subject_trigger = yes mam_censor_public_new_trigger = yes
    OR = { mam_censor_public_detention_trigger = yes
     AND = { mam_censor_public_trait_trigger = yes mam_censor_public_ground_trigger = yes }
    }
   } order_by = mam_censor_public_priority max = 1 check_range_bounds = no
    save_scope_as = mam_cen_public_target
    if = { limit = { mam_censor_public_detention_trigger = yes }
     save_scope_value_as = { name = mam_cen_public_kind value = 32 }
     imprisoner = { save_scope_as = mam_cen_public_accuser }
    }
    else = {
     save_scope_value_as = { name = mam_cen_public_kind value = 30 }
     title:h_china.holder = { save_scope_as = mam_cen_public_accuser }
     if = { limit = { NOT = { title:h_china.holder = { has_imprisonment_reason = scope:mam_cen_public_target } } }
      liege = { save_scope_as = mam_cen_public_accuser }
     }
    }
    mam_censor_public_capture_effect = yes
   }
  }
 }
}
# Accused character context. Source is an actual current public status, an
# explicit native-reason holder's referral, or the native on_imprison callback.
mam_censor_public_capture_effect = {
 save_scope_as = mam_cen_public_subject
 if = { limit = {
  mam_censor_public_subject_trigger = yes mam_censor_public_new_trigger = yes
  exists = scope:mam_cen_public_accuser exists = scope:mam_cen_public_kind
  exists = title:h_china.holder exists = title:e_minister_censor.holder
  title:h_china.holder = { mam_censor_emperor_trigger = yes }
  OR = {
   AND = { scope:mam_cen_public_kind = 30 mam_censor_public_trait_trigger = yes mam_censor_public_ground_trigger = yes }
   AND = { scope:mam_cen_public_kind = 31 scope:mam_cen_public_accuser = {
    is_alive = yes is_imprisoned = no has_imprisonment_reason = scope:mam_cen_public_subject top_liege = title:h_china.holder
   } }
   AND = { scope:mam_cen_public_kind = 32 mam_censor_public_detention_trigger = yes
    imprisoner = scope:mam_cen_public_accuser
   }
  }
  NOT = { title:h_china = { any_in_list = { variable = mam_cen_intel_queue count >= 8 always = yes } } }
 }
  clear_saved_scope = mam_cen_new_intel
  title:h_china.holder = { create_story = { type = mam_censor_intelligence_lead_story save_scope_as = mam_cen_new_intel } }
  if = { limit = { exists = scope:mam_cen_new_intel }
   scope:mam_cen_new_intel = {
    set_variable = { name = mam_cen_realm value = title:h_china }
    set_variable = { name = mam_cen_suspect value = scope:mam_cen_public_subject }
    set_variable = { name = mam_cen_discoverer value = scope:mam_cen_public_accuser }
    set_variable = { name = mam_cen_public_accuser value = scope:mam_cen_public_accuser }
    set_variable = { name = mam_cen_intel_kind value = scope:mam_cen_public_kind }
    set_variable = { name = mam_cen_identified value = 1 }
    mam_censor_intel_setup_effect = yes
   }
   set_variable = { name = mam_cen_public_record value = scope:mam_cen_new_intel }
   set_variable = { name = mam_cen_public_seen value = 1 }
   # Do not sanction an already resolved confidential murder again merely
   # because the same public murderer status appeared after exposure.
   if = { limit = { scope:mam_cen_public_kind = 30
    has_trait = murderer exists = var:mam_cen_last_inspection
    var:mam_cen_last_inspection = { has_variable = mam_cen_native_fact
     var:mam_cen_native_fact = 6
     OR = { has_variable = mam_cen_sanctioned exists = var:mam_cen_retired }
    }
   }
    scope:mam_cen_new_intel = { mam_censor_intel_finish_effect = yes }
   }
   debug_log = "MAM_CENSOR_PUBLIC: senior criminal matter recorded; allegation is not proof"
  }
 }
}
