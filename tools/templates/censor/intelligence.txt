mam_censor_intel_capture_secret_effect = {

 # sanctioned report. A buried unsanctioned report can still reach the emperor.
 if = { limit = { exists = var:mam_cen_intel_recorded
  var:mam_cen_intel_recorded = {
   OR = { var:mam_cen_intel_pending = 1
    AND = { exists = var:mam_cen_inspection
     var:mam_cen_inspection = { OR = { mam_censor_live_trigger = yes has_variable = mam_cen_sanctioned exists = var:mam_cen_retired } }
    }
   }
  }
 }
  if = { limit = { scope:mam_cen_intel_reader = title:h_china.holder }
   set_variable = { name = mam_cen_intel_imperial_recorded value = 1 }
  }
 }
 else = {
 clear_saved_scope = mam_cen_new_intel
 title:h_china.holder = { create_story = { type = mam_censor_intelligence_lead_story save_scope_as = mam_cen_new_intel } }
 if = { limit = { exists = scope:mam_cen_new_intel }
  secret_owner = { save_scope_as = mam_cen_capture_suspect }
  save_scope_value_as = { name = mam_cen_capture_kind value = 1 }
  if = { limit = { secret_type = secret_embezzler } save_scope_value_as = { name = mam_cen_capture_kind value = 2 } }
  else_if = { limit = { secret_type = secret_exam_cheater } save_scope_value_as = { name = mam_cen_capture_kind value = 3 } }
  else_if = { limit = { secret_type = secret_request_incursion } save_scope_value_as = { name = mam_cen_capture_kind value = 4 } }
  else_if = { limit = { secret_type = secret_coup_plotter } save_scope_value_as = { name = mam_cen_capture_kind value = 5 } }
  else_if = { limit = { secret_type = secret_murder } save_scope_value_as = { name = mam_cen_capture_kind value = 6 } }
  else_if = { limit = { secret_type = secret_murder_attempt } save_scope_value_as = { name = mam_cen_capture_kind value = 7 } }
  scope:mam_cen_new_intel = {
   set_variable = { name = mam_cen_realm value = title:h_china }
   set_variable = { name = mam_cen_suspect value = scope:mam_cen_capture_suspect }
   set_variable = { name = mam_cen_intel_kind value = scope:mam_cen_capture_kind }
   set_variable = { name = mam_cen_discoverer value = scope:mam_cen_intel_reader }

   mam_censor_intel_setup_effect = yes
  }
  secret_target ?= { scope:mam_cen_new_intel = { set_variable = { name = mam_cen_victim value = prev } } }

   set_variable = { name = mam_cen_intel_imperial_recorded value = 1 }
  }
  debug_log = "MAM_CENSOR_INTEL: recorded an actually known official secret"
 }
 }
