# Emperor scope. Realm cadence guard prevents duplicate scans after succession or reload.
mam_revenue_quarter_effect = {
 # legacy common-fund guard
 if = { limit = { title:h_china = { NOT = { has_variable = mam_cf_active } } }

 if = { limit = { mam_revenue_emperor_trigger = yes }
  mam_revenue_maintenance_effect = yes
  if = { limit = { title:h_china = { OR = { NOT = { has_variable = mam_rv_scan_until } var:mam_rv_scan_until <= current_date } }
   exists = title:e_minister_of_revenue.holder title:e_minister_of_revenue.holder = { mam_revenue_officer_trigger = yes }
  }
   save_scope_as = mam_rv_scan_emperor
   title:e_minister_of_revenue.holder = { save_scope_as = mam_rv_revenue }
   title:h_china = { set_variable = { name = mam_rv_scan_until value = { value = current_date add = 85 } } }
   title:h_china = { change_variable = { name = mam_rv_scan_count add = 1 } set_variable = { name = mam_rv_last_need_count value = 0 } }
   clear_variable_list = mam_rv_scan_candidates
   every_vassal_or_below = { limit = {
    mam_revenue_recipient_trigger = yes mam_revenue_needs_help_trigger = yes
    primary_title = { NOT = { has_variable = mam_rv_funding_file } mam_revenue_aid_available_trigger = yes
     trigger_if = { limit = { has_variable = mam_rv_petition_until } var:mam_rv_petition_until <= current_date }
    }
   }
    save_scope_as = mam_rv_scanned
    scope:mam_rv_scan_emperor = { add_to_variable_list = { name = mam_rv_scan_candidates target = scope:mam_rv_scanned } }
    title:h_china = { change_variable = { name = mam_rv_last_need_count add = 1 } }
   }
   ordered_in_list = { variable = mam_rv_scan_candidates order_by = mam_revenue_priority_score max = 2 check_range_bounds = no
    mam_revenue_open_aid_effect = yes
   }
   clear_variable_list = mam_rv_scan_candidates
   if = { limit = { title:h_china = { var:mam_rv_priority = 1 } }
    ordered_vassal_or_below = { limit = {
     mam_revenue_province_trigger = yes treasury >= { value = mam_revenue_account_reserve add = 100 }
     monthly_character_treasury_balance >= 0
     NOT = { mam_revenue_has_recovery_need_trigger = yes }
     NOT = { mam_revenue_conflicting_settlement_trigger = yes }
     primary_title = { NOT = { has_variable = mam_rv_settlement_file } NOT = { has_variable = mam_rv_funding_file }
      trigger_if = { limit = { has_variable = mam_rv_settlement_until } var:mam_rv_settlement_until <= current_date }
      trigger_if = { limit = { has_variable = mam_rv_donor_until } var:mam_rv_donor_until <= current_date }
      trigger_if = { limit = { has_variable = mam_rv_settlement_review_until } var:mam_rv_settlement_review_until <= current_date }
     }
    } order_by = treasury max = 1 check_range_bounds = no mam_revenue_open_settlement_effect = yes }
   }
   # Review one eligible departmental allocation when its finances support it.
   ordered_councillor = { limit = {
    mam_revenue_budget_office_trigger = yes
    primary_title = { NOT = { has_variable = mam_rv_budget_file }
     trigger_if = { limit = { has_variable = mam_rv_budget_review_until } var:mam_rv_budget_review_until <= current_date }
    }
    OR = {
     AND = { treasury < 100 mam_revenue_budget_level < 2 monthly_character_treasury_balance < 0 }
     AND = { treasury > 1500 mam_revenue_budget_level > 0 title:h_china = { var:mam_rv_priority = 1 } NOT = { mam_revenue_has_project_trigger = yes } }
    }
   } order_by = treasury max = 1 check_range_bounds = no
    mam_revenue_open_budget_effect = yes
   }
   debug_log = "MAM_REVENUE: quarterly needs scan completed; at most two new funding cases and one budget review"
   if = { limit = { title:h_china = { var:mam_rv_last_need_count = 0 } } debug_log = "MAM_REVENUE: no eligible new funding needs; policy/settlement/budget work considered separately" }
  }
  mam_revenue_wake_effect = yes
 }

 }
}

  }
