# Candidate scope: add only candidates passing destination-specific native gates.
mam_add_posting_candidate_effect = {
    if = {
        limit = {
            mam_eligible_for_office_trigger = { TITLE = scope:mam_posting }
            NOT = { is_in_list = mam_posting_candidates }
        }
        add_to_list = mam_posting_candidates
    }
}
