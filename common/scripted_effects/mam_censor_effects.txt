# All state is owned by an institutional story indexed on h_china.
# ROOT/actor contexts are deliberately not stored in Personnel or Revenue.
mam_censor_start_effect = {
 if = { limit = {
  exists = scope:mam_cen_subject exists = title:h_china.holder exists = title:e_minister_censor.holder
  OR = { title:e_minister_censor.holder = { mam_censor_officer_trigger = yes }
   AND = { mam_censor_emperor_trigger = yes mam_censor_intel_can_open_trigger = yes scope:mam_cen_subject = title:e_minister_censor.holder }
  }
  OR = {
   scope:mam_cen_subject = { mam_censor_subject_trigger = { EMPEROR = title:h_china.holder CENSOR = title:e_minister_censor.holder } }
   AND = { mam_censor_emperor_trigger = yes mam_censor_intel_can_open_trigger = yes
    scope:mam_cen_subject = title:e_minister_censor.holder }
   AND = { mam_censor_intel_can_open_trigger = yes
    scope:mam_cen_intel_lead.var:mam_cen_intel_kind >= 30
    scope:mam_cen_subject = { mam_censor_public_subject_trigger = yes }
   }
  }
  title:h_china = { NOT = { exists = var:mam_cen_active } }
 }
  clear_saved_scope = mam_cen_file
  title:h_china.holder = { create_story = { type = mam_censor_inspection_story save_scope_as = mam_cen_file } }
  # Native effect previews do not create a story. Do not dereference a phantom
  # file when the interaction/agenda tooltip is being built.
  if = { limit = { exists = scope:mam_cen_file }
  scope:mam_cen_file = {
   set_variable = { name = mam_cen_realm value = title:h_china }
   set_variable = { name = mam_cen_subject value = scope:mam_cen_subject }
   set_variable = { name = mam_cen_office value = scope:mam_cen_subject.primary_title }
   set_variable = { name = mam_cen_censor value = title:e_minister_censor.holder }
   set_variable = { name = mam_cen_pending value = 1 }
   set_variable = { name = mam_cen_created value = current_date }
   set_variable = { name = mam_cen_serial value = 0 }
   set_variable = { name = mam_cen_style value = 0 }
   set_variable = { name = mam_cen_document value = 0 }
   set_variable = { name = mam_cen_proof value = 0 }
   set_variable = { name = mam_cen_obstruction value = 0 }
   set_variable = { name = mam_cen_admission value = 0 }
   set_variable = { name = mam_cen_concealed value = 0 }
   set_variable = { name = mam_cen_abuse value = 0 }
   set_variable = { name = mam_cen_backing value = 0 }
   set_variable = { name = mam_cen_response value = 0 }
   set_variable = { name = mam_cen_legal value = 0 }
   set_variable = { name = mam_cen_returned value = 0 }
   set_variable = { name = mam_cen_connected_heard value = 0 }
   set_variable = { name = mam_cen_false_return value = 0 }
   set_variable = { name = mam_cen_return_exposed value = 0 }
   set_variable = { name = mam_cen_imperial_seen value = 0 }
   if = { limit = { exists = scope:mam_cen_originator }
    set_variable = { name = mam_cen_originator value = scope:mam_cen_originator }
   }
   mam_censor_read_sources_effect = yes
   mam_censor_move_effect = { STAGE = 1 DAYS = 1 }
   if = { limit = { exists = scope:mam_cen_intel_lead } mam_censor_intel_attach_effect = yes }
   if = { limit = { exists = scope:mam_cen_intel_lead
    scope:mam_cen_intel_lead = { has_variable = mam_cen_urgent }
   }
    set_variable = { name = mam_cen_author value = scope:mam_cen_originator }
    mam_censor_move_effect = { STAGE = 7 DAYS = 7 }
   }
   if = { limit = { var:mam_cen_subject = title:e_minister_censor.holder }
    set_variable = { name = mam_cen_recused value = 1 }
    mam_censor_move_effect = { STAGE = 7 DAYS = 1 }
   }
  }
  title:h_china = { set_variable = { name = mam_cen_active value = scope:mam_cen_file } }
  title:h_china.var:mam_cen_intel_brief ?= { mam_censor_intel_defer_effect = yes }
  scope:mam_cen_subject = { add_character_flag = { flag = mam_cen_recent_inspection years = 2 } }
  scope:mam_cen_subject = { set_variable = { name = mam_cen_last_inspection value = scope:mam_cen_file } }
  scope:mam_cen_file = { set_variable = { name = mam_cen_due value = current_date } mam_censor_tick_effect = yes }
  debug_log = "MAM_CENSOR: opened institutional inspection; no guilt or payment created"
  }
 }
}
# READ ONLY adapters. A patronage linkage is evidence of an exchange, not a
# generated criminal secret. A valid public contribution receipt is not theft.
mam_censor_read_sources_effect = {
 mam_justice_read_receipts_effect = yes
 save_scope_as = mam_cen_reading
 var:mam_cen_realm.holder = {
  every_owned_story = { limit = {
   save_temporary_scope_as = mam_cen_source_candidate
   story_type = mam_personnel_patronage_story
   exists = var:mam_patron_minister exists = var:mam_patron_payer
   var:mam_patron_minister = scope:mam_cen_reading.var:mam_cen_subject
   has_variable = mam_patron_appointment has_variable = mam_patron_appointed
   has_variable = mam_patron_amount has_variable = mam_patron_date
   var:mam_patron_appointed >= { value = current_date subtract = 1825 }
   scope:mam_cen_reading = { NOT = { exists = var:mam_cen_source } }
   NOT = { title:h_china = { any_in_list = { variable = mam_cen_reviewed_sources this = scope:mam_cen_source_candidate } } }
  }
   save_scope_as = mam_cen_source_record
   scope:mam_cen_reading = {
    set_variable = { name = mam_cen_source value = scope:mam_cen_source_record }
    set_variable = { name = mam_cen_document value = 1 }
    set_variable = { name = mam_cen_payment value = scope:mam_cen_source_record.var:mam_patron_amount }
    set_variable = { name = mam_cen_payment_date value = scope:mam_cen_source_record.var:mam_patron_date }
    set_variable = { name = mam_cen_appointment value = scope:mam_cen_source_record.var:mam_patron_appointment }
    set_variable = { name = mam_cen_connected value = scope:mam_cen_source_record.var:mam_patron_payer }
   }
  }
 }
 var:mam_cen_realm = { every_in_list = { variable = mam_pc_receipts limit = {
  exists = var:mam_pc_actor var:mam_pc_actor = scope:mam_cen_reading.var:mam_cen_subject
  scope:mam_cen_reading = { NOT = { has_variable = mam_cen_receipt_seen } }
 }
  scope:mam_cen_reading = { set_variable = { name = mam_cen_receipt_seen value = 1 } }
 } }
}
mam_censor_move_effect = {
 set_variable = { name = mam_cen_stage value = $STAGE$ }
 change_variable = { name = mam_cen_serial add = 1 }
 set_variable = { name = mam_cen_due value = { value = current_date add = $DAYS$ } }
 set_variable = { name = mam_cen_stage_since value = current_date }
 remove_variable = mam_cen_reviewer remove_variable = mam_cen_sent remove_variable = mam_cen_until
}
# Weekly institutional worker. Does not silently choose an unanswered human option.
mam_censor_tick_effect = {
 if = { limit = { mam_censor_live_trigger = yes }
  save_scope_as = mam_cen_file
  if = { limit = { OR = {
   NOT = { exists = var:mam_cen_subject } var:mam_cen_subject = { is_alive = no }
   var:mam_cen_subject = title:h_china.holder
   var:mam_cen_subject = { top_liege != title:h_china.holder }
   var:mam_cen_subject = { NOR = { tgp_is_any_minister = yes is_governor_or_admin_count = yes } }
   var:mam_cen_realm.holder = { mam_censor_emperor_trigger = no }
  } } mam_censor_close_effect = { RESULT = 10 } }
  else_if = { limit = { var:mam_cen_created <= { value = current_date subtract = 1095 } }
   mam_censor_close_effect = { RESULT = 10 }
  }
  else = {
   clear_saved_scope = mam_cen_next
   clear_saved_scope = mam_cen_current
   title:h_china.holder = { save_scope_as = mam_cen_emperor }
   title:e_minister_censor.holder ?= { if = { limit = { mam_censor_officer_trigger = yes } save_scope_as = mam_cen_current } }
   # Findings about the author/fabricator remain personal when the office changes.
   if = { limit = { exists = scope:mam_cen_current }
    if = { limit = { var:mam_cen_subject = scope:mam_cen_current }
     # A new Censor cannot inherit the right to judge their own inspection.
     set_variable = { name = mam_cen_recused value = 1 }
     if = { limit = { var:mam_cen_stage = 9 } mam_censor_close_effect = { RESULT = 6 } }
     else_if = { limit = { NOR = { var:mam_cen_stage = 7 var:mam_cen_stage = 10 } }
      mam_censor_move_effect = { STAGE = 7 DAYS = 7 }
     }
     clear_saved_scope = mam_cen_current
    }
    else = { set_variable = { name = mam_cen_censor value = scope:mam_cen_current } }
   }
   switch = { trigger = var:mam_cen_stage
    1 = { scope:mam_cen_current ?= { save_scope_as = mam_cen_next } }
    2 = { var:mam_cen_subject = { save_scope_as = mam_cen_next } }
    3 = {
     if = { limit = { NOT = { exists = var:mam_cen_petitioner } }
      var:mam_cen_subject = { save_scope_as = mam_cen_subject }
      scope:mam_cen_emperor = {
       ordered_vassal_or_below = { limit = {
        tgp_is_any_minister = yes is_alive = yes is_imprisoned = no
        this != scope:mam_cen_subject this != scope:mam_cen_current
       } order_by = mam_censor_defender_weight max = 1 check_range_bounds = no
        scope:mam_cen_file = { set_variable = { name = mam_cen_petitioner value = prev } }
       }
      }
     }
     var:mam_cen_petitioner ?= { if = { limit = { is_alive = yes is_imprisoned = no
      top_liege = scope:mam_cen_emperor this != scope:mam_cen_file.var:mam_cen_subject this != scope:mam_cen_current }
      save_scope_as = mam_cen_next
     } }
     if = { limit = { NOT = { exists = scope:mam_cen_next } }
      scope:mam_cen_emperor.cp:councillor_chancellor ?= { if = { limit = { is_alive = yes is_imprisoned = no
       this != scope:mam_cen_file.var:mam_cen_subject this != scope:mam_cen_current }
       save_scope_as = mam_cen_next
      } }
     }
     if = { limit = { NOT = { exists = scope:mam_cen_next } } mam_censor_move_effect = { STAGE = 4 DAYS = 7 } }
    }
    4 = { scope:mam_cen_current ?= { save_scope_as = mam_cen_next } }
    5 = {
     scope:mam_cen_emperor.cp:minister_justice ?= { if = { limit = { is_alive = yes is_imprisoned = no
      has_title = title:e_minister_of_justice this != scope:mam_cen_file.var:mam_cen_subject this != scope:mam_cen_current }
      save_scope_as = mam_cen_next
     } }
     if = { limit = { NOT = { exists = scope:mam_cen_next } } scope:mam_cen_emperor = { save_scope_as = mam_cen_next } }
    }
    6 = { scope:mam_cen_current ?= { save_scope_as = mam_cen_next } }
    7 = { scope:mam_cen_emperor = { save_scope_as = mam_cen_next } }
    8 = { var:mam_cen_connected ?= { if = { limit = { is_alive = yes is_imprisoned = no top_liege = scope:mam_cen_emperor }
     save_scope_as = mam_cen_next
    } }
     if = { limit = { NOT = { exists = scope:mam_cen_next } } mam_censor_move_effect = { STAGE = 6 DAYS = 7 } }
    }
    9 = { scope:mam_cen_current ?= { save_scope_as = mam_cen_next } }
    10 = { scope:mam_cen_emperor = { save_scope_as = mam_cen_next } }
    11 = { var:mam_cen_subject = { save_scope_as = mam_cen_next } }
    12 = { var:mam_cen_subject = { save_scope_as = mam_cen_next } }
   }
   if = { limit = { exists = scope:mam_cen_next OR = { exists = scope:mam_cen_current var:mam_cen_stage = 7 var:mam_cen_stage = 10 } }
    if = { limit = { OR = { NOT = { exists = var:mam_cen_reviewer } var:mam_cen_reviewer != scope:mam_cen_next } }
     remove_variable = mam_cen_sent remove_variable = mam_cen_until
     set_variable = { name = mam_cen_reviewer value = scope:mam_cen_next }
     change_variable = { name = mam_cen_serial add = 1 }
    }
    if = { limit = { has_variable = mam_cen_until var:mam_cen_until <= current_date } remove_variable = mam_cen_sent }
    if = { limit = { var:mam_cen_due <= current_date NOT = { has_variable = mam_cen_sent } }
     mam_censor_present_effect = yes
    }
   }
  }
 }
}

# Record inspection-era responses rather than assuming financial trouble is guilt.
mam_censor_examine_effect = {
 mam_justice_check_receipt_effect = yes
 if = { limit = { var:mam_cen_document = 1 }
  if = { limit = { var:mam_cen_concealed = 0 } set_variable = { name = mam_cen_proof value = 1 } }
  else = {
   save_scope_value_as = { name = mam_cen_payment_roll value = mam_censor_payment_detection_chance }
   set_variable = { name = mam_cen_last_payment_chance value = scope:mam_cen_payment_roll }
   random = { chance = { value = scope:mam_cen_payment_roll }
     set_variable = { name = mam_cen_proof value = 1 }
     set_variable = { name = mam_cen_obstruction value = 1 }
   }
  }
 }
 if = { limit = { var:mam_cen_document = 0 var:mam_cen_response = 1
  NOT = { has_variable = mam_cen_native_fact } NOT = { has_variable = mam_cen_public_case }
 } mam_censor_move_effect = { STAGE = 12 DAYS = 7 } }
 else = { mam_censor_move_effect = { STAGE = 3 DAYS = 14 } }
}

mam_censor_close_effect = {
 mam_justice_censor_closed_effect = yes
 if = { limit = { mam_censor_live_trigger = yes }
  set_variable = { name = mam_cen_pending value = 0 }
  set_variable = { name = mam_cen_result value = $RESULT$ }
  # No preview-only traversal into post-write state or result correspondence.
  if = { limit = { has_variable = mam_cen_result }
  set_variable = { name = mam_cen_closed value = current_date }
  if = { limit = {
   # set_variable does not run in effect preview. The result is absent there.
   trigger_if = { limit = { has_variable = mam_cen_result }
    OR = { var:mam_cen_result = 2 var:mam_cen_result = 3 var:mam_cen_result = 4 var:mam_cen_result = 5 }
   }
   trigger_else = { always = no }
   var:mam_cen_abuse = 0 mam_censor_supported_trigger = yes
   exists = var:mam_cen_author var:mam_cen_author != var:mam_cen_subject }
   var:mam_cen_author = { if = { limit = { is_alive = yes }
    add_character_modifier = { modifier = mam_cen_trusted years = 3 }
   } }
  }
  change_variable = { name = mam_cen_serial add = 1 }
  remove_variable = mam_cen_sent remove_variable = mam_cen_until
  if = { limit = { exists = var:mam_cen_source }
   var:mam_cen_realm = { add_to_variable_list = { name = mam_cen_reviewed_sources target = prev.var:mam_cen_source } }
  }
  mam_censor_bind_effect = yes
  var:mam_cen_realm = {
   if = { limit = { var:mam_cen_active = scope:mam_cen_file } remove_variable = mam_cen_active }
   set_variable = { name = mam_cen_next_auto value = { value = current_date add = 180 } }
  }
  # One final correspondence per affected human; no endless background popup.
  var:mam_cen_subject ?= { if = { limit = { is_alive = yes is_ai = no } mam_censor_notify_effect = yes } }
  var:mam_cen_censor ?= { if = { limit = { is_alive = yes is_ai = no this != scope:mam_cen_subject } mam_censor_notify_effect = yes } }
  var:mam_cen_retired ?= { if = { limit = { is_alive = yes is_ai = no this != scope:mam_cen_subject this != scope:mam_cen_censor } mam_censor_notify_effect = yes } }
  var:mam_cen_defender ?= { if = { limit = { is_alive = yes is_ai = no } mam_censor_notify_effect = yes } }
  var:mam_cen_challenger ?= { if = { limit = { is_alive = yes is_ai = no } mam_censor_notify_effect = yes } }
  var:mam_cen_legal_reviewer ?= { if = { limit = { is_alive = yes is_ai = no } mam_censor_notify_effect = yes } }
  if = { limit = { OR = {
   var:mam_cen_imperial_seen = 1
   AND = { exists = var:mam_cen_originator var:mam_cen_originator = var:mam_cen_realm.holder }
  } }
   var:mam_cen_realm.holder ?= { if = { limit = { is_ai = no this != scope:mam_cen_subject this != scope:mam_cen_censor
    trigger_if = { limit = { scope:mam_cen_file = { exists = var:mam_cen_retired } } this != scope:mam_cen_file.var:mam_cen_retired }
   } mam_censor_notify_effect = yes } }
  }
  debug_log = "MAM_CENSOR: closed once; no automatic crime, score or money created"
  }
 }
}

# Sovereign context, bounded to one new public referral per weekly intake.
mam_censor_public_poll_effect = {
 if = { limit = { mam_censor_emperor_trigger = yes }
  every_vassal_or_below = { limit = { mam_censor_public_subject_trigger = yes has_variable = mam_cen_public_seen
   is_imprisoned = no mam_censor_public_ground_trigger = no
  }
   # A genuinely cleared episode may later acquire new grounds. Never reopen
   # a pending inspection just because its arrest consumed the native reason.
   if = { limit = { NOT = { exists = var:mam_cen_public_record } }
    remove_variable = mam_cen_public_record remove_variable = mam_cen_public_seen
   }
   else_if = { limit = { var:mam_cen_public_record = { var:mam_cen_intel_pending = 0
    trigger_if = { limit = { exists = var:mam_cen_inspection } var:mam_cen_inspection = { mam_censor_live_trigger = no } }
   } }
    remove_variable = mam_cen_public_record remove_variable = mam_cen_public_seen
   }
  }
  if = { limit = { NOT = { title:h_china = { any_in_list = { variable = mam_cen_intel_queue count >= 8 always = yes } } }
   exists = title:e_minister_censor.holder
  }
   ordered_vassal_or_below = { limit = {
    mam_censor_public_subject_trigger = yes mam_censor_public_new_trigger = yes
    OR = { mam_censor_public_detention_trigger = yes
     AND = { mam_censor_public_trait_trigger = yes mam_censor_public_ground_trigger = yes }
    }
   } order_by = mam_censor_public_priority max = 1 check_range_bounds = no
    save_scope_as = mam_cen_public_target
    if = { limit = { mam_censor_public_detention_trigger = yes }
     save_scope_value_as = { name = mam_cen_public_kind value = 32 }
     imprisoner = { save_scope_as = mam_cen_public_accuser }
    }
    else = {
     save_scope_value_as = { name = mam_cen_public_kind value = 30 }
     title:h_china.holder = { save_scope_as = mam_cen_public_accuser }
     if = { limit = { NOT = { title:h_china.holder = { has_imprisonment_reason = scope:mam_cen_public_target } } }
      liege = { save_scope_as = mam_cen_public_accuser }
     }
    }
    mam_censor_public_capture_effect = yes
   }
  }
 }
}
# Accused character context. Source is an actual current public status, an
# explicit native-reason holder's referral, or the native on_imprison callback.
mam_censor_public_capture_effect = {
 save_scope_as = mam_cen_public_subject
 if = { limit = {
  mam_censor_public_subject_trigger = yes mam_censor_public_new_trigger = yes
  exists = scope:mam_cen_public_accuser exists = scope:mam_cen_public_kind
  exists = title:h_china.holder exists = title:e_minister_censor.holder
  title:h_china.holder = { mam_censor_emperor_trigger = yes }
  OR = {
   AND = { scope:mam_cen_public_kind = 30 mam_censor_public_trait_trigger = yes mam_censor_public_ground_trigger = yes }
   AND = { scope:mam_cen_public_kind = 31 scope:mam_cen_public_accuser = {
    is_alive = yes is_imprisoned = no has_imprisonment_reason = scope:mam_cen_public_subject top_liege = title:h_china.holder
   } }
   AND = { scope:mam_cen_public_kind = 32 mam_censor_public_detention_trigger = yes
    imprisoner = scope:mam_cen_public_accuser
   }
  }
  NOT = { title:h_china = { any_in_list = { variable = mam_cen_intel_queue count >= 8 always = yes } } }
 }
  clear_saved_scope = mam_cen_new_intel
  title:h_china.holder = { create_story = { type = mam_censor_intelligence_lead_story save_scope_as = mam_cen_new_intel } }
  if = { limit = { exists = scope:mam_cen_new_intel }
   scope:mam_cen_new_intel = {
    set_variable = { name = mam_cen_realm value = title:h_china }
    set_variable = { name = mam_cen_suspect value = scope:mam_cen_public_subject }
    set_variable = { name = mam_cen_discoverer value = scope:mam_cen_public_accuser }
    set_variable = { name = mam_cen_public_accuser value = scope:mam_cen_public_accuser }
    set_variable = { name = mam_cen_intel_kind value = scope:mam_cen_public_kind }
    set_variable = { name = mam_cen_identified value = 1 }
    mam_censor_intel_setup_effect = yes
   }
   set_variable = { name = mam_cen_public_record value = scope:mam_cen_new_intel }
   set_variable = { name = mam_cen_public_seen value = 1 }
   # Do not sanction an already resolved confidential murder again merely
   # because the same public murderer status appeared after exposure.
   if = { limit = { scope:mam_cen_public_kind = 30
    has_trait = murderer exists = var:mam_cen_last_inspection
    var:mam_cen_last_inspection = { has_variable = mam_cen_native_fact
     var:mam_cen_native_fact = 6
     OR = { has_variable = mam_cen_sanctioned exists = var:mam_cen_retired }
    }
   }
    scope:mam_cen_new_intel = { mam_censor_intel_finish_effect = yes }
   }
   debug_log = "MAM_CENSOR_PUBLIC: senior criminal matter recorded; allegation is not proof"
  }
 }
}
