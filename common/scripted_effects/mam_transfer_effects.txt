mam_add_replacement_effect = {
    if = {
        limit = { mam_replacement_eligible_trigger = yes NOT = { is_in_list = mam_replacements } }
        add_to_list = mam_replacements
    }
}
mam_prepare_nomination_effect = {
 mam_refresh_power_context_effect = { SUBJECT = scope:mam_inspected }
 if = { limit = { exists = scope:mam_service_case } scope:mam_incumbent ?= { mam_update_one_dossier_effect = yes } }
 if = { limit = { exists = scope:mam_service_case NOT = { mam_nomination_ready_trigger = yes } }
  trigger_event = mam_personnel.0065
 }

 if = { limit = { mam_nomination_ready_trigger = yes }

    clear_saved_scope = mam_source
    clear_saved_scope = mam_replacement
 clear_saved_scope = mam_replacement_1
 clear_saved_scope = mam_replacement_2
 clear_saved_scope = mam_replacement_3

    scope:mam_inspected = {
        save_scope_as = mam_nominee
        if = { limit = { is_landed = yes } primary_title = { save_scope_as = mam_source } }
    }
    if = {
        limit = { exists = scope:mam_source }
        scope:mam_emperor = {
            every_courtier_or_guest = { mam_add_replacement_effect = yes }
            every_vassal_or_below = {
                # Landless noble-family heads are realm vassals, not courtiers.
                mam_add_replacement_effect = yes
                every_courtier_or_guest = { mam_add_replacement_effect = yes }
            }
        }

        ordered_in_list = {
            list = mam_replacements
            limit = {  }
            order_by = mam_recommendation_score
            max = 1
            check_range_bounds = no
            save_scope_as = mam_replacement_1
        }
        if = { limit = { exists = scope:mam_replacement_1 }

        ordered_in_list = {
            list = mam_replacements
            limit = { this != scope:mam_replacement_1 }
            order_by = mam_recommendation_score
            max = 1
            check_range_bounds = no
            save_scope_as = mam_replacement_2
        }
        }
        if = { limit = { exists = scope:mam_replacement_2 }

        ordered_in_list = {
            list = mam_replacements
            limit = { this != scope:mam_replacement_1 this != scope:mam_replacement_2 }
            order_by = mam_recommendation_score
            max = 1
            check_range_bounds = no
            save_scope_as = mam_replacement_3
        }
        }

    }
    if = {
        limit = { is_ai = yes }
        scope:mam_emperor.primary_title = {
            set_variable = { name = mam_ai_last_stage value = 4 }
            if = {
                limit = { OR = { NOT = { exists = scope:mam_source } exists = scope:mam_replacement_1 } }
                set_variable = { name = mam_ai_last_stage value = 5 }
            }
        }
    }
    trigger_event = mam_personnel.0051

 }
}

# Queue a snapshot of personally held constituent titles and jurisdiction vassals.
# TITLE/OLD/NEW are supplied explicitly. No broad resignation helper is called.
mam_queue_bundle_effect = {
    every_in_list = { list = mam_bundle_parts remove_from_list = mam_bundle_parts }
    $OLD$ = {
        every_held_title = {
            limit = {
                this != $TITLE$
                is_landless_type_title = no
                is_noble_family_title = no
                tier >= tier_county
                target_is_de_jure_liege_or_above = $TITLE$
            }
            add_to_list = mam_bundle_parts
        }
    }
    every_in_list = {
        list = mam_bundle_parts
        limit = { holder = $OLD$ target_is_de_jure_liege_or_above = $TITLE$ }
        change_title_holder_include_vassals = {
            holder = $NEW$
            change = scope:mam_appointment_change
            take_baronies = no
        }
    }
    $TITLE$ = {
        change_title_holder_include_vassals = {
            holder = $NEW$
            change = scope:mam_appointment_change
            take_baronies = no
        }
    }
}

# Single resolution avoids an intentional landless interval for a transferring player.
# Runtime verification is required: script execution has no general rollback facility.
# A stale event must never dismiss a newer proposal's lock.
mam_clear_matching_proposal_effect = {
    # Title variables remain supported after a character dies. Match both the
    # emperor reference and generation so an old timer cannot clear new business.
    scope:mam_proposal_realm ?= {
        if = {
            limit = {
                exists = scope:mam_generation
                has_variable = mam_active_proposal_generation
                var:mam_active_proposal_generation = scope:mam_generation
                has_variable = mam_active_proposal_emperor
                var:mam_active_proposal_emperor = scope:mam_emperor
            }
            remove_variable = mam_active_proposal_generation
            remove_variable = mam_active_proposal_emperor
        }
    }
    scope:mam_emperor ?= {
        if = {
            limit = {
                is_alive = yes
                exists = scope:mam_generation
                has_variable = mam_proposal_generation
                var:mam_proposal_generation = scope:mam_generation
            }
            remove_character_flag = mam_pending_proposal
            set_variable = { name = mam_resolved_generation value = scope:mam_generation }
        }
    }
}

        }
    }
}

# Annual AI opportunity review: lowest-scoring supported office; three-year review cooldown.
# The realm-title opt-in survives succession. Stage counters are QA diagnostics.
mam_ai_review_effect = {
    mam_resolve_register_authority_effect = yes
    scope:mam_emperor = {
        if = { limit = { NOT = { has_variable = mam_ai_reviews_attempted } } set_variable = { name = mam_ai_reviews_attempted value = 0 } }
        change_variable = { name = mam_ai_reviews_attempted add = 1 }
        primary_title = {
            if = { limit = { NOT = { has_variable = mam_ai_realm_reviews } } set_variable = { name = mam_ai_realm_reviews value = 0 } }
            change_variable = { name = mam_ai_realm_reviews add = 1 }
            set_variable = { name = mam_ai_last_stage value = 0 }
        }
    }
    add_character_flag = { flag = mam_ai_review_cooldown years = 3 }
    mam_select_service_case_effect = yes
    if = { limit = { NOT = { exists = scope:mam_service_case } }
    scope:mam_emperor = {
        ordered_vassal = {
            limit = {
                primary_title = { NOT = { mam_service_case_ai_cooldown_trigger = yes } }
                mam_power_title_allowed_trigger = { TITLE = primary_title }
                mam_supported_governor_trigger = yes
                mam_clean_bundle_trigger = { TITLE = primary_title }
                NOT = { has_character_flag = mam_recent_appointment }
            }
            order_by = { value = mam_recommendation_score multiply = -1 }
            max = 1
            check_range_bounds = no
            save_scope_as = mam_incumbent
            primary_title = { save_scope_as = mam_posting }
        }
    }
    }
    if = {
        limit = { exists = scope:mam_posting }
        save_scope_value_as = { name = mam_review_mode value = 2 }
        mam_build_posting_review_effect = yes
        scope:mam_incumbent = {
            save_scope_value_as = { name = mam_incumbent_score value = mam_recommendation_score }
        }
        scope:mam_emperor.primary_title = {
            set_variable = { name = mam_ai_last_stage value = 1 }
            set_variable = { name = mam_ai_last_incumbent_score value = scope:mam_incumbent_score }
        }
        ordered_in_list = {
            list = mam_posting_candidates
            limit = {
                mam_posting_eligible_trigger = yes
                NOT = { has_character_flag = mam_recent_appointment }
                mam_public_suitability >= scope:mam_incumbent.mam_public_suitability
                mam_recommendation_score >= { value = scope:mam_incumbent_score add = 15 }
                trigger_if = {
                    limit = { is_landed = yes }
                    mam_clean_source_bundle_trigger = { TITLE = primary_title }
                }
            }
            order_by = mam_recommendation_score
            max = 1
            check_range_bounds = no
            save_scope_as = mam_inspected
        }
        mam_try_private_nominee_effect = yes
        if = {
            limit = { exists = scope:mam_inspected mam_nomination_ready_trigger = yes }
            scope:mam_emperor.primary_title = {
                set_variable = { name = mam_ai_last_stage value = 3 }
                set_variable = { name = mam_ai_last_nominee_score value = scope:mam_inspected.mam_recommendation_score }
            }
            scope:mam_emperor.primary_title = {
                if = { limit = { NOT = { has_variable = mam_ai_qualified_selections } } set_variable = { name = mam_ai_qualified_selections value = 0 } }
                change_variable = { name = mam_ai_qualified_selections add = 1 }
            }
            mam_prepare_nomination_effect = yes
        }
    }
}
