# Caller binds subject title; all records are owned by the living Chinese emperor.
# SLOT: funding/budget/settlement/annual/mandate/priority. SLOT_ID: 1..6.
mam_revenue_create_file_effect = {
 clear_saved_scope = mam_rv_created
 if = { limit = { exists = scope:mam_rv_subject }
  scope:mam_rv_subject = { if = { limit = { has_variable = mam_rv_$SLOT$_file OR = {
   NOT = { exists = var:mam_rv_$SLOT$_file }
   AND = { exists = var:mam_rv_$SLOT$_file var:mam_rv_$SLOT$_file = { mam_revenue_file_current_trigger = no } }
  } } remove_variable = mam_rv_$SLOT$_file } }
 }
 if = { limit = { exists = scope:mam_rv_subject title:h_china.holder = { mam_revenue_emperor_trigger = yes }
  scope:mam_rv_subject = { NOT = { has_variable = mam_rv_$SLOT$_file } }
 }
  title:h_china.holder = { create_story = { type = mam_revenue_case_story save_scope_as = mam_rv_created } }
  # Tooltip simulation does not create a story or execute set_variable effects.
  # Do not inspect a story until it actually exists.
  if = { limit = { exists = scope:mam_rv_created }
  title:h_china = {
   if = { limit = { NOT = { has_variable = mam_rv_serial } } set_variable = { name = mam_rv_serial value = 0 } }
   change_variable = { name = mam_rv_serial add = 1 }
   add_to_variable_list = { name = mam_rv_files target = scope:mam_rv_created }
  }
  scope:mam_rv_created = {
   set_variable = { name = mam_rv_subject value = scope:mam_rv_subject }
   set_variable = { name = mam_rv_slot value = $SLOT_ID$ }
   set_variable = { name = mam_rv_kind value = $KIND$ }
   set_variable = { name = mam_rv_stage value = 0 }
   set_variable = { name = mam_rv_pending value = 1 }
   set_variable = { name = mam_rv_created value = current_date }
   set_variable = { name = mam_rv_position value = title:h_china.var:mam_rv_serial }
   set_variable = { name = mam_rv_generation value = 0 }
   set_variable = { name = mam_rv_amount value = 100 }
   set_variable = { name = mam_rv_urgency value = 20 }
   set_variable = { name = mam_rv_reserved value = 1 }
   set_variable = { name = mam_rv_donor_title value = title:h_china.holder.primary_title }
   set_variable = { name = mam_rv_recipient_title value = scope:mam_rv_subject }
   if = { limit = { exists = title:e_minister_of_revenue.holder } set_variable = { name = mam_rv_proposer value = title:e_minister_of_revenue.holder } }
  }
  scope:mam_rv_subject = { set_variable = { name = mam_rv_$SLOT$_file value = scope:mam_rv_created } }
  debug_log = "MAM_REVENUE: created a deduplicated fiscal record"
  }
 }
}
mam_revenue_release_effect = {
 save_scope_as = mam_rv_releasing
 remove_variable = mam_rv_presented
 var:mam_rv_actor ?= { if = { limit = { exists = var:mam_rv_open_file var:mam_rv_open_file = scope:mam_rv_releasing } remove_variable = mam_rv_open_file } }
 remove_variable = mam_rv_actor
 title:h_china.holder ?= { trigger_event = { id = mam_revenue.99 days = 2 } }
}

}
mam_revenue_close_effect = {
 save_scope_as = mam_rv_closing
 set_variable = { name = mam_rv_pending value = 0 }
 set_variable = { name = mam_rv_closed value = current_date }
 if = { limit = { var:mam_rv_slot = 1 } mam_revenue_clear_slot_effect = { SLOT = funding } }
 else_if = { limit = { var:mam_rv_slot = 2 } mam_revenue_clear_slot_effect = { SLOT = budget } }
 else_if = { limit = { var:mam_rv_slot = 3 } mam_revenue_clear_slot_effect = { SLOT = settlement } }
 else_if = { limit = { var:mam_rv_slot = 4 } mam_revenue_clear_slot_effect = { SLOT = annual } }
 else_if = { limit = { var:mam_rv_slot = 5 } mam_revenue_clear_slot_effect = { SLOT = mandate } }
 else_if = { limit = { var:mam_rv_slot = 6 } mam_revenue_clear_slot_effect = { SLOT = priority } }
 else_if = { limit = { var:mam_rv_slot = 7 } mam_revenue_clear_slot_effect = { SLOT = appeal } }
 title:h_china = { remove_list_variable = { name = mam_rv_files target = scope:mam_rv_closing } }
 mam_revenue_release_effect = yes
 # Keep a short factual outcome record; no judicial runtime or indefinite story growth.
}
