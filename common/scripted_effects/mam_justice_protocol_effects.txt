mam_justice_observe_case_effect = {
 # An independently executed native action closes obsolete custom proposals;
 # no cost or native reward is replayed. Censor/Personnel callbacks remain native.
 if = { limit = { exists = scope:recipient.var:mam_jus_case }
  scope:recipient.var:mam_jus_case = {
   save_scope_as = mam_jus_observed_case
   if = { limit = { has_variable = mam_jus_pending var:mam_jus_pending = 1 NOT = { has_variable = mam_jus_executing } }
    set_variable = { name = mam_jus_last_performer value = scope:puppet_or_actor }
    set_variable = { name = mam_jus_last_action value = $ACTION$ }
    set_variable = { name = mam_jus_paid_treasury value = 0 }
    set_variable = { name = mam_jus_paid_gold value = 0 }
    set_variable = { name = mam_jus_paid_influence value = 0 }
    set_variable = { name = mam_jus_paid_prestige value = 0 }
    set_variable = { name = mam_jus_paid_piety value = 0 }
    set_variable = { name = mam_jus_native_external value = 1 }
    if = { limit = { $ACTION$ = mam_justice_action_1 }
     set_variable = { name = mam_jus_arrest_done value = 1 } mam_justice_move_effect = { STAGE = 4 DAYS = 30 }
    }
    else_if = { limit = { $ACTION$ = mam_justice_action_2 } mam_justice_close_effect = { RESULT = 2 } }
    else_if = { limit = { $ACTION$ = mam_justice_action_3 }
     if = { limit = { exists = scope:banish scope:banish = yes } mam_justice_close_effect = { RESULT = 5 } }
     else = { mam_justice_close_effect = { RESULT = 3 } }
    }
    else_if = { limit = { $ACTION$ = mam_justice_action_4 }
     if = { limit = { var:mam_jus_subject = { is_alive = no } } mam_justice_close_effect = { RESULT = 4 } }
     else = {
      set_variable = { name = mam_jus_performer value = scope:puppet_or_actor }
      set_variable = { name = mam_jus_pending_death value = 1 }
      set_variable = { name = mam_jus_death_until value = { value = current_date add = 90 } }
      mam_justice_move_effect = { STAGE = 13 DAYS = 7 }
     }
    }
   }
  }
 }
