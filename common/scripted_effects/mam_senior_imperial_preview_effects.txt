# Emperor scope. Preview the consequences for the actual nominee/sponsor/recommender.
# Never reuse Personnel's root-based stress preview on the emperor.
mam_senior_imperial_approval_preview_effect = {
 if = { limit = { exists = scope:mam_senior_nominee }
 scope:mam_senior_nominee = {
  if = { limit = { is_alive = no } custom_tooltip = mam_senior.block_dead_tt }
  if = { limit = { is_adult = no } custom_tooltip = mam_senior.block_age_tt }
  if = { limit = { is_imprisoned = yes } custom_tooltip = mam_senior.block_prison_tt }
  if = { limit = { top_liege.primary_title != scope:mam_senior_file.var:mam_senior_realm } custom_tooltip = mam_senior.block_realm_tt }
  if = { limit = { merit_level < scope:mam_senior_file.var:mam_senior_merit_floor } custom_tooltip = mam_senior.block_merit_tt }
  if = { limit = { any_held_title = { tier > scope:mam_posting.tier is_landless_type_title = no is_noble_family_title = no } } custom_tooltip = mam_senior.block_rank_tt }
  if = { limit = { mam_senior_choice_trigger = yes mam_senior_source_actionable_trigger = no }
   custom_tooltip = mam_senior.imperial_source_blocked_tt
  }
 }
 if = { limit = { mam_senior_confirmation_actor_trigger = yes scope:mam_senior_nominee = { mam_senior_choice_trigger = yes mam_senior_source_actionable_trigger = yes } }
  custom_tooltip = mam_senior.imperial_approve_tt
  scope:mam_senior_nominee = {
   if = { limit = { mam_senior_first_office_trigger = yes } custom_tooltip = mam_senior.imperial_first_tt }
   else_if = { limit = { mam_senior_lateral_trigger = yes } custom_tooltip = mam_senior.imperial_lateral_tt }
   else_if = { limit = { mam_senior_promotion_trigger = yes } custom_tooltip = mam_senior.imperial_promotion_tt }
   if = { limit = { has_character_modifier = mam_personnel_service_endorsement }
    custom_tooltip = mam_senior.consequence_endorsement
    show_as_tooltip = { remove_character_modifier = mam_personnel_service_endorsement }
   }
  }
  if = { limit = { exists = scope:mam_senior_file.var:mam_senior_sponsored_candidate
   scope:mam_senior_file.var:mam_senior_sponsored_candidate = scope:mam_senior_nominee
   NOT = { scope:mam_senior_file = { has_variable = mam_senior_credit_awarded } }
  }
   mam_senior_sponsorship_preview_effect = { CANDIDATE = scope:mam_senior_nominee SPONSOR = scope:mam_senior_file.var:mam_senior_sponsor }
  }
  scope:mam_senior_file.var:mam_senior_recommender ?= {
   if = { limit = { mam_can_nominate_trigger = yes }
    if = { limit = { scope:mam_senior_file = { var:mam_senior_personal_recommendation = 1 }
     OR = { has_trait = just has_trait = honest }
     exists = scope:mam_senior_file.var:mam_senior_professional
     scope:mam_senior_nominee.mam_public_suitability < { value = scope:mam_senior_file.var:mam_senior_professional.mam_public_suitability subtract = 5 }
    }
     custom_tooltip = mam_senior.consequence_stress_deferred
     show_as_tooltip = { stress_impact = { just = 15 honest = 15 } }
    }
    if = { limit = { has_trait = vengeful scope:mam_senior_nominee = { has_relation_rival = prev } }
     custom_tooltip = mam_senior.consequence_stress_deferred
     show_as_tooltip = { stress_impact = { vengeful = 15 } }
    }
    if = { limit = { exists = scope:mam_senior_nominee.var:mam_patronage_record
     scope:mam_senior_nominee.var:mam_patronage_record = { var:mam_patron_minister = prev }
     scope:mam_senior_nominee = { has_variable = mam_bribe_until var:mam_bribe_until > current_date }
    } custom_tooltip = mam_senior.consequence_obligation }
   }
  }
 }
 }
}
