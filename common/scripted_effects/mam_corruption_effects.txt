# Family ties are facts, not a guilty verdict. Payment is kept as private evidence.

 scope:mam_posting = {
  set_variable = { name = mam_case_minister value = scope:mam_minister }
  set_variable = { name = mam_case_nominee value = scope:mam_nominee }
  set_variable = { name = mam_case_emperor value = scope:mam_emperor }
  set_variable = { name = mam_case_filed value = current_date }
  set_variable = { name = mam_case_generation value = scope:mam_generation }
  set_variable = { name = mam_case_public_score value = scope:mam_nominee.mam_public_suitability }
  set_variable = { name = mam_case_qualification value = scope:mam_nominee.mam_qualification_points }
  set_variable = { name = mam_case_merit_rank value = scope:mam_nominee.merit_level }
  set_variable = { name = mam_case_best_public_score value = scope:mam_power_display_best }
  set_variable = { name = mam_case_ranked_law value = 0 }

  if = { limit = { scope:mam_emperor = { has_realm_law = mam_discretion_ranked } } set_variable = { name = mam_case_ranked_law value = 1 } }
  if = { limit = { scope:mam_emperor = { has_realm_law = mam_recruitment_strict } } set_variable = { name = mam_case_strict_exam_law value = 1 } }
  remove_variable = mam_case_source
  remove_variable = mam_case_source_replacement
  if = { limit = { exists = scope:mam_source } set_variable = { name = mam_case_source value = scope:mam_source } }
  if = { limit = { exists = scope:mam_replacement } set_variable = { name = mam_case_source_replacement value = scope:mam_replacement } }

  set_variable = { name = mam_case_kin value = 0 }
  set_variable = { name = mam_case_friend value = 0 }
  set_variable = { name = mam_case_house value = 0 }
  set_variable = { name = mam_case_political value = 0 }

  if = { limit = { scope:mam_nominee = { has_relation_friend = scope:mam_minister } } set_variable = { name = mam_case_friend value = 1 } }

  if = { limit = { scope:mam_nominee = { mam_political_client_trigger = yes } } set_variable = { name = mam_case_political value = 1 } }
 }
