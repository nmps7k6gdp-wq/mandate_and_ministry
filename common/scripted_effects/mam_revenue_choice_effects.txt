  custom_tooltip = mam_rv.native_credit_tt
  if = { limit = { scope:mam_rv_file = { has_variable = mam_rv_native_hook } } custom_tooltip = mam_rv.native_hook_tt }
  if = { limit = { scope:mam_rv_file = { has_variable = mam_rv_clear_debt } } custom_tooltip = mam_rv.debt_hook_tt }
 }

 }
}
