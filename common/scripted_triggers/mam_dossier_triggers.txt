# Title scope. Realm identity is a title, not the emperor's character identity.
mam_dossier_record_current_trigger = {
 has_variable = mam_ds_holder has_variable = mam_ds_realm
 has_variable = mam_ds_started has_variable = mam_ds_sampled
 has_variable = mam_ds_baseline_count
 NOT = { has_variable = mam_ds_invalidated }
 holder = var:mam_ds_holder
 var:mam_ds_realm = holder.top_liege.primary_title
 var:mam_ds_baseline_count > 0
}
