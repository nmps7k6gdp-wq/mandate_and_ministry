mam_revenue_fund_native_start_location_8_trigger = {

				text = great_project_type_construct_great_barracks_military_governor_duchy_capital
				barony.county.holder ?= {
					vassal_contract_has_flag = celestial_military_appointment # is held by a military governor

					title_capital_county.title_province = scope:province # is also the duchy capital
				}
			}
			custom_tooltip = {
				text = great_project_type_construct_great_barracks_does_not_already_have_barracks
				NOT = {
					has_building = celestial_great_barracks_01 # you can overwrite duchy buildings, but there's no point in doing it to the same one
				}
			}
			trigger_if = {
				limit = {
					barony.county.holder = {
						is_ai = no
						this != root
					}
				}
				custom_tooltip = {
					text = great_project_type_construct_great_barracks_do_not_override_player_building
					NOT = {
						has_building_with_flag = duchy_building
					}
				}
			}
		}

mam_revenue_fund_native_start_location_17_trigger = {

		scope:province.county = {
			holder.top_liege ?= {
				custom_description = {
					text = neither_at_war_or_truce
					subject = scope:founder
					object = this
					NOR = {
						has_truce = scope:founder
						is_at_war_with = scope:founder
					}
				}
			}
			trigger_if = { # Unreformed faiths are always fair game, organized ones only during the Conversion phase
				limit = {
					NOT = { faith = { has_doctrine_parameter = unreformed } }
				}
				trigger_if = {
					limit = {
						scope:founder = {
							any_character_situation = {
								situation_type = the_christian_church
							}
						}
					}
					custom_tooltip = {
						text = the_christian_church_mass_rite_conversion_decision_tt
						pam_mass_rite_conversion_conversion_phase_trigger = yes
					}
				}
				trigger_else = {
					custom_tooltip = {
						text = pam_mass_rite_conversion_target_unreformed_faith
						always = no
					}
				}
				custom_tooltip = {
					text = pam_mass_rite_conversion_target_organized_faith
					faith != scope:founder.faith
				}
			}
		}
		custom_tooltip = {
			text = pam_mass_rite_conversion_already_targets_province
			NOT = {
				any_great_project = {
					great_project_type = pam_mass_rite_conversion
					any_in_list = {
						variable = mass_converted_counties
						this = scope:province.county
					}
				}
			}
		}
		trigger_if = {
			limit = {
				scope:founder = {
					is_ai = yes
				}
			}
			scope:province.county.faith = {
				faith_hostility_level = {
					target = scope:founder.faith
					value >= faith_hostile_level
				}
			}
		}
	
}
