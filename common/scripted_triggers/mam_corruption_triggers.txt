# Bribe payer. AI decision resolves the active minister freshly from the top liege.
mam_can_offer_bribe_trigger = {
 trigger_if = { limit = { exists = top_liege.cp:minister_personnel }
 has_tgp_dlc_trigger = yes is_alive = yes is_adult = yes gold >= 100
 is_independent_ruler = no this != top_liege.cp:minister_personnel
 NOT = { has_character_flag = mam_bribe_offer_cooldown }
 NOT = { AND = { has_variable = mam_bribe_until has_variable = mam_bribe_sponsor var:mam_bribe_until > current_date var:mam_bribe_sponsor = top_liege.cp:minister_personnel } }
 top_liege = {
  government_has_flag = government_is_celestial
  cp:minister_personnel = {
   is_alive = yes is_imprisoned = no mam_cj_office_allowed_trigger = yes has_title = title:e_minister_of_personnel
   NOT = { has_variable = mam_bribe_offer_sender }
   NOT = { has_character_flag = mam_bribe_inbox_cooldown }
  }
 }
}
 trigger_else = { always = no }
}
