# Candidate in the benchmark pool. The raw native adapter never calls discretion.
mam_power_raw_candidate_trigger = {
 this != scope:mam_incumbent
 mam_eligible_for_office_trigger = { TITLE = scope:mam_power_target }
 trigger_if = { limit = { scope:mam_power_replacements_only = 1 }
  is_landed = no this != scope:mam_power_subject
 }
 trigger_else = {
  trigger_if = { limit = { is_landed = yes }
   mam_clean_source_bundle_trigger = { TITLE = primary_title }
   mam_power_title_allowed_trigger = { TITLE = primary_title }
   save_temporary_scope_as = mam_power_pool_subject
   primary_title = { save_temporary_scope_as = mam_power_pool_source }
   scope:mam_emperor = {
    OR = {
     any_courtier_or_guest = { mam_power_pool_replacement_trigger = yes }
     any_vassal_or_below = {
      OR = { mam_power_pool_replacement_trigger = yes any_courtier_or_guest = { mam_power_pool_replacement_trigger = yes } }
     }
    }
   }
  }
 }
}
mam_power_pool_replacement_trigger = {
 is_landed = no this != scope:mam_power_pool_subject this != scope:mam_incumbent
 mam_eligible_for_office_trigger = { TITLE = scope:mam_power_pool_source }
}
