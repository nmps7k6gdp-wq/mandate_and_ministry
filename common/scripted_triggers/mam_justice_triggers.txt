mam_justice_native_commit_trigger = {
 OR = {
  $ACTOR$ = { NOT = { has_title = title:e_minister_of_justice } }
  $TARGET$ = { mam_justice_tracked_trigger = no }
  AND = { exists = scope:mam_jus_case
   scope:mam_jus_case = { mam_justice_live_trigger = yes has_variable = mam_jus_executing var:mam_jus_executing = 1
    var:mam_jus_performer = $ACTOR$ var:mam_jus_subject = $TARGET$
    OR = { var:mam_jus_action = $ACTION$ AND = { var:mam_jus_action = 5 $ACTION$ = mam_justice_action_3 } }
   }
  }
  AND = {
   $ACTOR$ = { mam_justice_officer_trigger = yes NOT = { has_character_modifier = mam_jus_restrained } }
   title:h_china.holder = { NOT = { has_realm_law = mam_justice_authority_reserved } }
   $TARGET$ = { mam_justice_person_trigger = yes this != $ACTOR$ tgp_is_any_minister = no
    NOT = { is_close_or_extended_family_of = title:h_china.holder }
    trigger_if = { limit = { exists = var:mam_jus_case } var:mam_jus_case = { NOT = { has_variable = mam_jus_supervised } } }
   }
   OR = {
    AND = { $ACTION$ = mam_justice_action_1 $TARGET$ = { mam_justice_native_ground_trigger = yes } }
    AND = { $ACTION$ = mam_justice_action_2
     $TARGET$ = { exists = liege liege = { mam_justice_any_reason_trigger = { TARGET = $TARGET$ } } }
     NOT = { $TARGET$ = { exists = liege liege = { has_extra_criminal_reason_against_trigger = { CRIMINAL = $TARGET$ } } } }
    }
    AND = { $ACTION$ = mam_justice_action_3
     scope:banish = no scope:demand_conversion = no scope:renounce_claims = no scope:gain_hook = no
     scope:take_vows = no scope:recruit = no scope:disfigure = no scope:blind = no scope:castrate = no
     scope:demand_admin = no scope:become_executioner = no scope:change_prison = no scope:make_puppet = no
    }
   }
  }
 }
}
