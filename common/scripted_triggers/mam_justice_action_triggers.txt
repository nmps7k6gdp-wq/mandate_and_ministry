  exists = scope:mam_jus_pardon_authority scope:mam_jus_pardon_authority = { mam_justice_any_reason_trigger = { TARGET = scope:mam_jus_subject } }
  scope:mam_jus_performer = { is_character_interaction_valid = { recipient = scope:mam_jus_subject interaction = pardon_interaction } }
