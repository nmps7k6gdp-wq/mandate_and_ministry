# Highest distinction only; do not sum each exam victory.
mam_exam_distinction_points = {
 value = 0
 if = { limit = { mam_exam_triple_first_trigger = yes } add = 16 }
 else_if = { limit = { mam_exam_palace_first_trigger = yes } add = 12 }
 else_if = { limit = { mam_exam_palace_honors_trigger = yes } add = 8 }
 else_if = { limit = { has_character_flag = mam_exam_metropolitan_first } add = 6 }
 else_if = { limit = { has_character_flag = mam_exam_provincial_first mam_has_ordinary_provincial_evidence_trigger = yes } add = 4 }
}

  scope:mam_minister = { OR = { has_trait = just has_trait = honest has_trait = diligent has_trait = patient } }

}
